POLICY devolution_items.devolution_items_insert
  cmd=INSERT permissive=PERMISSIVE roles={authenticated}
  WITH CHECK : (EXISTS ( SELECT 1
   FROM devolutions d
  WHERE ((d.id = devolution_items.devolution_id) AND has_store_access(d.store_id))))

POLICY devolution_items.devolution_items_select
  cmd=SELECT permissive=PERMISSIVE roles={authenticated}
  USING      : (EXISTS ( SELECT 1
   FROM devolutions d
  WHERE ((d.id = devolution_items.devolution_id) AND has_store_access(d.store_id))))

POLICY devolutions.devolutions_insert_own_store
  cmd=INSERT permissive=PERMISSIVE roles={authenticated}
  WITH CHECK : has_store_access(store_id)

POLICY devolutions.devolutions_select_own_store
  cmd=SELECT permissive=PERMISSIVE roles={authenticated}
  USING      : has_store_access(store_id)

POLICY devolutions.devolutions_tenant_insert_with_check
  cmd=INSERT permissive=PERMISSIVE roles={authenticated}
  WITH CHECK : (store_id = ANY (current_user_store_ids()))

  cmd=UPDATE permissive=PERMISSIVE roles={authenticated}
  USING      : has_store_access(store_id)
